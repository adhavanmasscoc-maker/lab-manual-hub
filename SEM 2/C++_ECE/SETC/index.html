SET C - Q1: Single Linked List ADT

1. AIM

To implement a Singly Linked List ADT with basic insertion and display operations.

3. C++ PROGRAM

#include <iostream>
using namespace std;

struct Node
{
    int data;
    Node *nextNode;
};

class SinglyLinkedList
{
    Node *headNode = NULL;
public:
    void insertNode(int value)
    {
        Node *newNode = new Node;
        newNode->data = value;
        newNode->nextNode = NULL;

        if(headNode == NULL)
        {
            headNode = newNode;
        }
        else
        {
            Node *tempNode = headNode;
            while(tempNode->nextNode != NULL)
            {
                tempNode = tempNode->nextNode;
            }
            tempNode->nextNode = newNode;
        }
    }

    void displayList()
    {
        Node *tempNode = headNode;
        while(tempNode != NULL)
        {
            cout << tempNode->data << "->";
            tempNode = tempNode->nextNode;
        }
        cout << "NULL\n";
    }
};

int main()
{
    SinglyLinkedList list;
    list.insertNode(10);
    list.insertNode(20);
    list.displayList();
    return 0;
}

#include <iostream>
using namespace std;

struct Node
{
    int coefficient;
    int power;
    Node *nextNode;
};

void insertTerm(Node*& headNode, int coeff, int pow)
{
    Node *newNode = new Node;
    newNode->coefficient = coeff;
    newNode->power = pow;
    newNode->nextNode = NULL;

    if(headNode == NULL)
    {
        headNode = newNode;
    }
    else
    {
        Node *tempNode = headNode;
        while(tempNode->nextNode != NULL)
        {
            tempNode = tempNode->nextNode;
        }
        tempNode->nextNode = newNode;
    }
}

void subtractPolynomials(Node *poly1, Node *poly2)
{
    while(poly1 != NULL && poly2 != NULL)
    {
        if(poly1->power > poly2->power)
        {
            cout << poly1->coefficient << "x^" << poly1->power << " ";
            poly1 = poly1->nextNode;
        }
        else if(poly1->power < poly2->power)
        {
            cout << -poly2->coefficient << "x^" << poly2->power << " ";
            poly2 = poly2->nextNode;
        }
        else
        {
            int resultCoeff = poly1->coefficient - poly2->coefficient;
            if(resultCoeff != 0)
            {
                cout << resultCoeff << "x^" << poly1->power << " ";
            }
            poly1 = poly1->nextNode;
            poly2 = poly2->nextNode;
        }
    }
    cout << "\n";
}

int main()
{
    Node *polynomial1 = NULL;
    Node *polynomial2 = NULL;
    
    insertTerm(polynomial1, 5, 2);
    insertTerm(polynomial1, 4, 1);
    
    insertTerm(polynomial2, 3, 2);
    insertTerm(polynomial2, 2, 0);
    
    subtractPolynomials(polynomial1, polynomial2);
    return 0;

SET C - Q8: Infix to Postfix Expression using Stack

1. AIM

To convert an Infix expression to Postfix using a Stack.

3. C++ PROGRAM

#include <iostream>
#include <stack>
using namespace std;

int getPrecedence(char operatorChar)
{
    if(operatorChar == '^')
    {
        return 3;
    }
    if(operatorChar == '*' || operatorChar == '/')
    {
        return 2;
    }
    if(operatorChar == '+' || operatorChar == '-')
    {
        return 1;
    }
    return -1;
}

void infixToPostfix(string expression)
{
    stack<char> characterStack;
    string resultString;
    
    for(char currentCharacter : expression)
    {
        if(isalnum(currentCharacter))
        {
            resultString += currentCharacter;
        }
        else if(currentCharacter == '(')
        {
            characterStack.push(currentCharacter);
        }
        else if(currentCharacter == ')')
        {
            while(characterStack.top() != '(')
            {
                resultString += characterStack.top();
                characterStack.pop();
            }
            characterStack.pop();
        }
        else
        {
            while(!characterStack.empty() && getPrecedence(currentCharacter) <= getPrecedence(characterStack.top()))
            {
                resultString += characterStack.top();
                characterStack.pop();
            }
            characterStack.push(currentCharacter);
        }
    }
    
    while(!characterStack.empty())
    {
        resultString += characterStack.top();
        characterStack.pop();
    }
    cout << resultString << "\n";
}

int main()
{
    infixToPostfix("A+B*C");
    return 0;
}

SET C - Q11: AVL Tree Insertion & Deletion

1. AIM

To implement AVL Tree self-balancing operations.

3. C++ PROGRAM

#include <iostream>
using namespace std;

struct AVLNode
{
    int dataValue;
    int height;
    AVLNode *leftChild;
    AVLNode *rightChild;
};

int getHeight(AVLNode *node)
{
    if(node != NULL)
    {
        return node->height;
    }
    return 0;
}

int getMax(int a, int b)
{
    if(a > b)
    {
        return a;
    }
    return b;
}

AVLNode* rotateRight(AVLNode *yNode)
{
    AVLNode *xNode = yNode->leftChild;
    yNode->leftChild = xNode->rightChild;
    xNode->rightChild = yNode;
    
    yNode->height = getMax(getHeight(yNode->leftChild), getHeight(yNode->rightChild)) + 1;
    xNode->height = getMax(getHeight(xNode->leftChild), getHeight(xNode->rightChild)) + 1;
    
    return xNode;
}

AVLNode* rotateLeft(AVLNode *xNode)
{
    AVLNode *yNode = xNode->rightChild;
    xNode->rightChild = yNode->leftChild;
    yNode->leftChild = xNode;
    
    xNode->height = getMax(getHeight(xNode->leftChild), getHeight(xNode->rightChild)) + 1;
    yNode->height = getMax(getHeight(yNode->leftChild), getHeight(yNode->rightChild)) + 1;
    
    return yNode;
}

int getBalanceFactor(AVLNode *node)
{
    if(node != NULL)
    {
        return getHeight(node->leftChild) - getHeight(node->rightChild);
    }
    return 0;
}

AVLNode* insertNode(AVLNode *node, int key)
{
    if(node == NULL)
    {
        AVLNode *newNode = new AVLNode;
        newNode->dataValue = key;
        newNode->height = 1;
        newNode->leftChild = NULL;
        newNode->rightChild = NULL;
        return newNode;
    }
    
    if(key < node->dataValue)
    {
        node->leftChild = insertNode(node->leftChild, key);
    }
    else
    {
        node->rightChild = insertNode(node->rightChild, key);
    }
    
    node->height = 1 + getMax(getHeight(node->leftChild), getHeight(node->rightChild));
    int balance = getBalanceFactor(node);
    
    if(balance > 1 && key < node->leftChild->dataValue)
    {
        return rotateRight(node);
    }
    if(balance < -1 && key > node->rightChild->dataValue)
    {
        return rotateLeft(node);
    }
    
    return node;
}

void printInOrder(AVLNode *rootNode)
{
    if(rootNode != NULL)
    {
        printInOrder(rootNode->leftChild);
        cout << rootNode->dataValue << " ";
        printInOrder(rootNode->rightChild);
    }
}

int main()
{
    AVLNode *root = NULL;
    root = insertNode(root, 10);
    root = insertNode(root, 20);
    root = insertNode(root, 30);
    
    printInOrder(root);
    return 0;
}

SET C - Q14: Balanced Parentheses using Stack

To check if parentheses in an expression are balanced.

3. C++ PROGRAM

#include <iostream>
#include <stack>
using namespace std;

bool isValidExpression(string expressionString)
{
    stack<char> bracketStack;
    
    for(char currentCharacter : expressionString)
    {
        if(currentCharacter == '(' || currentCharacter == '{' || currentCharacter == '[')
        {
            bracketStack.push(currentCharacter);
        }
        else
        {
            if(bracketStack.empty())
            {
                return false;
            }
            
            char topCharacter = bracketStack.top();
            bracketStack.pop();
            
            if((currentCharacter == ')' && topCharacter != '(') || 
               (currentCharacter == '}' && topCharacter != '{') || 
               (currentCharacter == ']' && topCharacter != '['))
            {
                return false;
            }
        }
    }
    return bracketStack.empty();
}

int main()
{
    if(isValidExpression("{[]}"))
    {
        cout << "Balanced\n";
    }
    else
    {
        cout << "Not Balanced\n";
    }
    return 0;
}

#include <iostream>
#include <vector>
using namespace std;

class MinimumHeap
{
    vector<int> heapArray;
public:
    void pushValue(int insertedValue)
    {
        heapArray.push_back(insertedValue);
        int currentIndex = heapArray.size() - 1;
        
        while(currentIndex != 0 && heapArray[(currentIndex - 1) / 2] > heapArray[currentIndex])
        {
            swap(heapArray[currentIndex], heapArray[(currentIndex - 1) / 2]);
            currentIndex = (currentIndex - 1) / 2;
        }
    }
    
    void displayHeap()
    {
        for(int heapElement : heapArray)
        {
            cout << heapElement << " ";
        }
        cout << "\n";
    }
};

SET C - Q18: Polynomial Addition using Singly Linked List

1. AIM

To add two polynomials represented as linked lists.

3. C++ PROGRAM

#include <iostream>
using namespace std;

struct PolynomialNode
{
    int coefficient;
    int exponent;
    PolynomialNode *nextNode;
};

void insertTerm(PolynomialNode*& headNode, int coeff, int exp)
{
    PolynomialNode *newNode = new PolynomialNode;
    newNode->coefficient = coeff;
    newNode->exponent = exp;
    newNode->nextNode = NULL;

    if(headNode == NULL)
    {
        headNode = newNode;
    }
    else
    {
        PolynomialNode *tempNode = headNode;
        while(tempNode->nextNode != NULL)
        {
            tempNode = tempNode->nextNode;
        }
        tempNode->nextNode = newNode;
    }
}

void addPolynomials(PolynomialNode *polyOne, PolynomialNode *polyTwo)
{
    while(polyOne != NULL && polyTwo != NULL)
    {
        if(polyOne->exponent > polyTwo->exponent)
        {
            cout << polyOne->coefficient << "x^" << polyOne->exponent << " + ";
            polyOne = polyOne->nextNode;
        }
        else if(polyOne->exponent < polyTwo->exponent)
        {
            cout << polyTwo->coefficient << "x^" << polyTwo->exponent << " + ";
            polyTwo = polyTwo->nextNode;
        }
        else
        {
            int sumCoefficient = polyOne->coefficient + polyTwo->coefficient;
            cout << sumCoefficient << "x^" << polyOne->exponent << " + ";
            
            polyOne = polyOne->nextNode;
            polyTwo = polyTwo->nextNode;
        }
    }
    cout << "0\n";
}

int main()
{
    PolynomialNode *firstPoly = NULL;
    PolynomialNode *secondPoly = NULL;
    
    insertTerm(firstPoly, 5, 2);
    insertTerm(firstPoly, 4, 1);
    
    insertTerm(secondPoly, 3, 2);
    insertTerm(secondPoly, 2, 0);
    
    addPolynomials(firstPoly, secondPoly);
    return 0;
}
