SET B - Q1: Single Linked List ADT Operations

1. AIM

To implement a Singly Linked List ADT with basic insertion and display operations.

3. C++ PROGRAM

#include <iostream>
using namespace std;

struct Node
{
    int data;
    Node *nextNode;
};

class SinglyLinkedList
{
    Node *headNode = NULL;
public:
    void insertNode(int value)
    {
        Node *newNode = new Node;
        newNode->data = value;
        newNode->nextNode = NULL;

        if(headNode == NULL)
        {
            headNode = newNode;
        }
        else
        {
            Node *tempNode = headNode;
            while(tempNode->nextNode != NULL)
            {
                tempNode = tempNode->nextNode;
            }
            tempNode->nextNode = newNode;
        }
    }

    void displayList()
    {
        Node *tempNode = headNode;
        while(tempNode != NULL)
        {
            cout << tempNode->data << "->";
            tempNode = tempNode->nextNode;
        }
        cout << "NULL\n";
    }
};

int main()
{
    SinglyLinkedList list;
    list.insertNode(10);
    list.insertNode(20);
    list.displayList();
    return 0;
}

#include <iostream>
using namespace std;

struct PolynomialNode
{
    int coefficient;
    int exponent;
    PolynomialNode *nextNode;
};

void insertTerm(PolynomialNode*& headNode, int coeff, int exp)
{
    PolynomialNode *newNode = new PolynomialNode;
    newNode->coefficient = coeff;
    newNode->exponent = exp;
    newNode->nextNode = NULL;

    if(headNode == NULL)
    {
        headNode = newNode;
    }
    else
    {
        PolynomialNode *tempNode = headNode;
        while(tempNode->nextNode != NULL)
        {
            tempNode = tempNode->nextNode;
        }
        tempNode->nextNode = newNode;
    }
}

void subtractPolynomials(PolynomialNode *polyOne, PolynomialNode *polyTwo)
{
    while(polyOne != NULL && polyTwo != NULL)
    {
        if(polyOne->exponent > polyTwo->exponent)
        {
            cout << polyOne->coefficient << "x^" << polyOne->exponent << " ";
            polyOne = polyOne->nextNode;
        }
        else if(polyOne->exponent < polyTwo->exponent)
        {
            cout << -polyTwo->coefficient << "x^" << polyTwo->exponent << " ";
            polyTwo = polyTwo->nextNode;
        }
        else
        {
            int resultCoeff = polyOne->coefficient - polyTwo->coefficient;
            if(resultCoeff != 0)
            {
                cout << resultCoeff << "x^" << polyOne->exponent << " ";
            }
            polyOne = polyOne->nextNode;
            polyTwo = polyTwo->nextNode;
        }
    }
    cout << "\n";
}

int main()
{
    PolynomialNode *firstPolynomial = NULL;
    PolynomialNode *secondPolynomial = NULL;
    
    insertTerm(firstPolynomial, 5, 2);
    insertTerm(firstPolynomial, 4, 1);
    
    insertTerm(secondPolynomial, 3, 2);
    insertTerm(secondPolynomial, 2, 0);
    
    subtractPolynomials(firstPolynomial, secondPolynomial);
    return 0;
}

            return;

        Node *tempNode = rearNode->nextNode;
        do
        {
            cout << tempNode->data << " ";
            tempNode = tempNode->nextNode;

SET B - Q8: Balanced Parentheses using Stack

1. AIM

To check if parentheses in an expression are balanced.

3. C++ PROGRAM

#include <iostream>
#include <stack>
using namespace std;

bool isValidExpression(string expressionString)
{
    stack<char> bracketStack;
    
    for(char currentCharacter : expressionString)
    {
        if(currentCharacter == '(' || currentCharacter == '{' || currentCharacter == '[')
        {
            bracketStack.push(currentCharacter);
        }
        else
        {
            if(bracketStack.empty())
            {
                return false;
            }
            
            char topCharacter = bracketStack.top();
            bracketStack.pop();
            
            if((currentCharacter == ')' && topCharacter != '(') || 
               (currentCharacter == '}' && topCharacter != '{') || 
               (currentCharacter == ']' && topCharacter != '['))
            {
                return false;
            }
        }
    }
    return bracketStack.empty();
}

int main()
{
    if(isValidExpression("{[]}"))
    {
        cout << "Balanced\n";
    }
    else
    {
        cout << "Not Balanced\n";
    }
    return 0;
}

SET B - Q11: Infix to Postfix Expression

1. AIM

To convert an Infix expression to Postfix using a Stack.

3. C++ PROGRAM

#include <iostream>
#include <stack>
using namespace std;

int getPrecedence(char operatorChar)
{
    if(operatorChar == '^')
    {
        return 3;
    }
    if(operatorChar == '*' || operatorChar == '/')
    {
        return 2;
    }
    if(operatorChar == '+' || operatorChar == '-')
    {
        return 1;
    }
    return -1;
}

void infixToPostfix(string expressionString)
{
    stack<char> characterStack;
    string resultString;
    
    for(char currentCharacter : expressionString)
    {
        if(isalnum(currentCharacter))
        {
            resultString += currentCharacter;
        }
        else if(currentCharacter == '(')
        {
            characterStack.push(currentCharacter);
        }
        else if(currentCharacter == ')')
        {
            while(characterStack.top() != '(')
            {
                resultString += characterStack.top();
                characterStack.pop();
            }
            characterStack.pop();
        }
        else
        {
            while(!characterStack.empty() && getPrecedence(currentCharacter) <= getPrecedence(characterStack.top()))
            {
                resultString += characterStack.top();
                characterStack.pop();
            }
            characterStack.push(currentCharacter);
        }
    }
    
    while(!characterStack.empty())
    {
        resultString += characterStack.top();
        characterStack.pop();
    }
    cout << resultString << "\n";
}

int main()
{
    infixToPostfix("A+B*C");
    return 0;
}

#include <iostream>
#include <vector>
using namespace std;

struct BTreeNode
{
    vector<int> nodeKeys;
    vector<BTreeNode*> childNodes;
    bool isLeafNode = true;
};

bool searchBTree(BTreeNode *rootNode, int searchKey)
{
    if(rootNode == NULL)
    {
        return false;
    }
    
    int index = 0;
    while(index < rootNode->nodeKeys.size() && searchKey > rootNode->nodeKeys[index])
    {
        index++;
    }
    
    if(index < rootNode->nodeKeys.size() && searchKey == rootNode->nodeKeys[index])
    {
        return true;
    }
    
    if(rootNode->isLeafNode)
    {
        return false;
    }
    
    return searchBTree(rootNode->childNodes[index], searchKey);
}

int main()
{
    BTreeNode *rootNode = new BTreeNode();
    rootNode->nodeKeys = {10, 20};
    
    if(searchBTree(rootNode, 20))
    {
        cout << "Found\n";
    }

#include <iostream>
#include <vector>
using namespace std;

class MaximumHeap
{
    vector<int> heapArray;
public:
    void insertValue(int newValue)
    {
        heapArray.push_back(newValue);
        int currentIndex = heapArray.size() - 1;
        
        while(currentIndex != 0 && heapArray[(currentIndex - 1) / 2] < heapArray[currentIndex])
        {
            swap(heapArray[currentIndex], heapArray[(currentIndex - 1) / 2]);
            currentIndex = (currentIndex - 1) / 2;
        }
    }
    
    void displayHeap()
    {
        for(int element : heapArray)
        {
            cout << element << " ";
        }
        cout << "\n";
    }
};

SET B - Q15: AVL Tree Insertion Operation

1. AIM

To implement AVL Tree self-balancing insertion.

3. C++ PROGRAM

#include <iostream>
using namespace std;

struct AVLNode
{
    int dataValue;
    int nodeHeight;
    AVLNode *leftChild;
    AVLNode *rightChild;
};

int getHeight(AVLNode *currentNode)
{
    if(currentNode != NULL)
    {
        return currentNode->nodeHeight;
    }
    return 0;
}

int getMaximum(int valueA, int valueB)
{
    if(valueA > valueB)
    {
        return valueA;
    }
    return valueB;
}

AVLNode* rotateRight(AVLNode *yNode)
{
    AVLNode *xNode = yNode->leftChild;
    yNode->leftChild = xNode->rightChild;
    xNode->rightChild = yNode;
    
    yNode->nodeHeight = getMaximum(getHeight(yNode->leftChild), getHeight(yNode->rightChild)) + 1;
    xNode->nodeHeight = getMaximum(getHeight(xNode->leftChild), getHeight(xNode->rightChild)) + 1;
    
    return xNode;
}

AVLNode* rotateLeft(AVLNode *xNode)
{
    AVLNode *yNode = xNode->rightChild;
    xNode->rightChild = yNode->leftChild;
    yNode->leftChild = xNode;
    
    xNode->nodeHeight = getMaximum(getHeight(xNode->leftChild), getHeight(xNode->rightChild)) + 1;
    yNode->nodeHeight = getMaximum(getHeight(yNode->leftChild), getHeight(yNode->rightChild)) + 1;
    
    return yNode;
}

int getBalanceFactor(AVLNode *currentNode)
{
    if(currentNode != NULL)
    {
        return getHeight(currentNode->leftChild) - getHeight(currentNode->rightChild);
    }
    return 0;
}

AVLNode* insertAVLNode(AVLNode *currentNode, int insertKey)
{
    if(currentNode == NULL)
    {
        AVLNode *newNode = new AVLNode;
        newNode->dataValue = insertKey;
        newNode->nodeHeight = 1;
        newNode->leftChild = NULL;
        newNode->rightChild = NULL;
        return newNode;
    }
    
    if(insertKey < currentNode->dataValue)
    {
        currentNode->leftChild = insertAVLNode(currentNode->leftChild, insertKey);
    }
    else
    {
        currentNode->rightChild = insertAVLNode(currentNode->rightChild, insertKey);
    }
    
    currentNode->nodeHeight = 1 + getMaximum(getHeight(currentNode->leftChild), getHeight(currentNode->rightChild));
    int currentBalance = getBalanceFactor(currentNode);
    
    if(currentBalance > 1 && insertKey < currentNode->leftChild->dataValue)
    {
        return rotateRight(currentNode);
    }
    if(currentBalance < -1 && insertKey > currentNode->rightChild->dataValue)
    {
        return rotateLeft(currentNode);
    }
    
    return currentNode;
}

void printInOrderTraversal(AVLNode *rootNode)
{
    if(rootNode != NULL)
    {
        printInOrderTraversal(rootNode->leftChild);
        cout << rootNode->dataValue << " ";
        printInOrderTraversal(rootNode->rightChild);
    }
}

int main()
{
    AVLNode *treeRoot = NULL;
    treeRoot = insertAVLNode(treeRoot, 10);
    treeRoot = insertAVLNode(treeRoot, 20);
    treeRoot = insertAVLNode(treeRoot, 30);
    
    printInOrderTraversal(treeRoot);
    return 0;
}
